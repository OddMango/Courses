### Ubuntu terminal Cheat Sheet for Git - Day 1 ###
# Create a branch and add a recipe
git switch -c new-recipe         # Create and switch to a new branch
nano sides/mixed-nuts.md         # Create a new recipe
git status                       # Check working tree
git add sides/mixed-nuts.md      # Stage the new recipe
git commit -m "Add recipe"       # Commit the new recipe

# Modify the recipe and commit again
nano sides/mixed-nuts.md         # Edit the recipe
git diff                         # View uncommitted changes
git add sides/mixed-nuts.md      # Stage changes
git commit -m "Update recipe"    # Commit changes

# Switch to main and modify a recipe
git switch main                  # Switch to main branch
git branch                       # Check current branch
nano sides/EXISTING-RECIPE.md    # Edit an existing recipe (replace filename)
git add sides/EXISTING-RECIPE.md # Stage modified recipe
git commit -m "Update recipe"    # Commit changes on main

# Browse commit history
git log --all                    # View full commit history
git log --oneline --all          # View compact commit history
git log --graph --oneline --all  # Visualize branches and commits
git show COMMIT                  # Show changes in one commit

#  Compare branches
git diff main new-recipe         # Compare changes between branches
git diff --stat main new-recipe  # Show summary of differences
git diff --name-only main new-recipe # List changed files
# GitHub: Compare branches

# Compare two arbitrary commits
git log --oneline --all          # Find commit hashes
git diff COMMIT1 COMMIT2         # Compare two commits
git show COMMIT1                 # Inspect first commit
git show COMMIT2                 # Inspect second commit
git diff a1e2324 4afef12        # Compare your paprika commits

# Rename a branch
git branch -m new-recipe better-recipe # Rename local branch
git branch                       # Verify branch name
git log --graph --oneline --all  # View updated history
git push -u origin better-recipe # Publish renamed branch

# Create a tag and GitHub release
git switch better-recipe         # Switch to recipe branch
git tag -a v1.0 -m "First version" # Create annotated tag
git tag                          # List tags
git show v1.0                    # Inspect tagged commit
git push origin v1.0             # Publish tag to GitHub

# Additional commands I tested
git branch                       # List local branches
git branch -a                    # List local and remote branches
git remote -v                    # Check connected remote repository
git add .                        # Stage all changes
git log -5                       # Show last five commits
git push -u origin BRANCH        # Publish branch to GitHub
git pull                         # Fetch and integrate remote changes
git switch main                  # Switch back to main
git switch -                     # Switch to previous branch
git tag -d TAG                   # Delete a local tag
